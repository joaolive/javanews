com.joaolive.javanews.user
│
├── domain/                              # 1. O NÚCLEO (Sem dependências de framework)
│   ├── User.java                        # A Entidade Rica de Domínio
│   ├── records/                         # Value Objects
│   │   ├── Bio.java
│   │   ├── Name.java
│   │   └── Username.java
│   ├── exception/                       # Exceções puras de negócio
│   │   ├── InvalidUsernameException.java
│   │   └── UserNotFoundException.java
│   └── ports/                           # Contratos (Portas)
│       ├── in/                          # O que a Aplicação oferece para o mundo externo
│       │   └── CreateUserUseCase.java   
│       └── out/                         # O que o Domínio precisa do mundo externo
│           └── UserRepository.java      # Interface do repositório (não tem JPA aqui!)
│
├── application/                         # 2. CASOS DE USO (Orquestração de regras)
│   ├── service/
│   │   └── CreateUserService.java       # Implementa CreateUserUseCase e chama o UserRepository
│   └── dto/                             # Objetos de transferência de dados internos (se necessários)
│       └── UserDto.java
│
└── infrastructure/                      # 3. ADAPTADORES (Frameworks, Banco, Web)
    ├── persistence/                     # Adaptadores de Banco de Dados (JPA/Hibernate)
    │   ├── UserJpaEntity.java           # A classe anêmica com @Entity
    │   ├── UserRepositoryAdapter.java   # Implementa domain.ports.out.UserRepository
    │   └── SpringDataJpaUserRepository.java # A interface do Spring Data (extends JpaRepository)
    │
    ├── web/                             # Adaptadores de Entrada (REST API)
    │   ├── UserController.java          # Recebe requisições HTTP e chama os UseCases (ports.in)
    │   └── dto/                         # DTOs específicos da Web (Payloads)
    │       ├── CreateUserRequest.java
    │       └── UserResponse.java
    │
    └── config/                          # Configurações do módulo (Beans, Injeção)
        └── UserModuleConfig.java        # Onde você instancia seus UseCases injetando os Adapters
